#include <iostream>

using namespace std;

int main()

{

    const int N = 5;

    int a[N][N];

    cout << "Введите элементы массива 5x5:\n";

    for (int i = 0; i < N; i++)

    {

        for (int j = 0; j < N; j++)

        {

            cin >> a[i][j];

        }

    }

    int min = a[0][0];

    for (int i = 0; i < N; i++)

    {

        for (int j = 0; j < N; j++)

        {

            if (a[i][j] < min)

            {

                min = a[i][j];

            }

        }

    }

    cout << "\nМинимальный элемент: " << min << endl;

    return 0;
}
